// ===================================================
// قاعدة أمان (Security Rule) لازم تتضاف في Firebase Console
// عشان ميزة "الشاتات المخفية" تشتغل.
//
// روح على: Firebase Console → Firestore Database → Rules
// وضيف الجزء ده جوه match /databases/{database}/documents { ... }
// (بجانب باقي الـ rules الموجودة عندك، من غير ما تمسح حاجة).
// ===================================================

match /hiddenChatsAuth/{userEmail} {
  // كل مستخدم يقدر يقرا/يكتب بس في المستند بتاعه هو (بريده الإلكتروني
  // هو الـ document ID). المستخدم لازم يكون مسجل دخول (حتى لو anonymous
  // auth، زي باقي التطبيق).
  allow read, write: if request.auth != null;
}

// ملحوظة: الـ rule دي بسيطة ومطابقة لنفس مستوى الأمان المستخدم في باقي
// أجزاء التطبيق (auth != null بس، من غير ربط صارم بالإيميل نفسه)، عشان
// تتوافق مع أسلوب المشروع الحالي. لو حابب تشديد الحماية أكتر بعدين
// (مثلاً التأكد إن userEmail == إيميل المستخدم المسجل فعليًا)، قولي
// وهساعدك تظبطها.
